## Global Constraints

- 依据 [设计文档](../specs/2026-09-12-toolbox-design.md) §2.2/§7 与 M1+M2 已落地契约（见「既有接口契约」节）。
- **火山 ASR WS 二进制协议**（官方文档 6561/1354869，已核实）：
  - 接口 `wss://openspeech.bytedance.com/api/v3/sauc/bigmodel_nostream`；鉴权 headers：`X-Api-App-Key`(APP ID)/`X-Api-Access-Key`(Token)/`X-Api-Resource-Id`/`X-Api-Request-Id`(UUID)/`X-Api-Sequence: -1`。Resource-Id 默认 `volc.bigasr.sauc.duration`，可配置。
  - 帧 = header(4B) + payload size(4B, 大端 uint32) + payload；全部整数大端。
  - header byte0 = `0x11`（version=0b0001, header size=0b0001，即 4 字节）；byte1 高 4 位 message type（`0b0001` full client request / `0b0010` audio only / `0b1001` full server response / `0b1111` server error），低 4 位 flags（`0b0000` 无 sequence / `0b0010` 最后一包无 sequence / `0b0011` 最后一包带负 sequence）；byte2 高 4 位 serialization（`0b0001` JSON / `0b0000` 无）、低 4 位 compression（`0b0001` Gzip）；byte3 = 0x00 保留。
  - full client request payload = Gzip(JSON)；audio 帧 payload = Gzip(音频)；nostream 模式音频**一次性单帧发送**，flags=`0b0010`（最后一包）；JSON 帧序列化=JSON+Gzip（byte2=0x11），音频帧序列化=无+Gzip（byte2=0x01）。
  - server response：header + sequence(4B，flags 低位为 1 时存在) + size(4B) + payload(Gzip JSON)；error 帧：header + error code(4B) + size(4B) + 错误 JSON。客户端读到 message type `0b1001` 且 flags 指示最后一包的响应为止（或 error 帧）。
  - 请求 JSON：`{"user":{"uid":"toolbox"},"audio":{"format":"mp3|wav|pcm|ogg","rate":16000,"bits":16,"channel":1,"language":"zh-CN"},"request":{"model_name":"bigmodel","enable_itn":true,"show_utterances":true,"result_type":"full"}}`（audio 的 rate/bits/channel 按 format 为 mp3/ogg 时由服务端自解析，仍按文档传 16000/16/1）。
  - 响应 JSON：`{"audio_info":{"duration":<ms>},"result":{"text":"全文","utterances":[{"text","start_time":<ms>,"end_time":<ms>,"definite":true,"words":[...]}]}}`。
- **异步通道**（官方文档 6561/1354868，M2 已核实）：`POST https://openspeech.bytedance.com/api/v3/auc/bigmodel/submit`（body `{"audio_url":...}`）→ `POST .../bigmodel/query` 轮询；headers 同上 + `X-Api-Sequence: -1`（仅 submit）；Resource-Id `volc.seedasr.auc`；响应 `{"id","status":"Completed|Running|Queuing|Failed"...}`（实现时以响应实际字段为准解析 text 与 utterances，utterances 缺失则跳过 SRT）。
- **REST 统一包络**（用户硬性要求）：HTTP 一律 200，`{code,data,message}`；码 0/2/3/4/5/6；用 `internal/server/apierr.go` 的 ok()/fail()/failErr()。stream/download 二进制流例外。WS 消息不套包络。
- **既有接口契约（M1+M2 已落地，必须复用不得重构）**：

```go
package volcengine

import (
	"bytes"
	"compress/gzip"
	"encoding/binary"
	"fmt"
)

// 火山 ASR WS 二进制协议分帧（大端）。参考官方文档 6561/1354869。
const (
	msgTypeFullRequest  = 0b0001
	msgTypeAudioOnly    = 0b0010
	msgTypeFullResponse = 0b1001
	msgTypeError        = 0b1111

	flagNoSeq      = 0b0000
	flagLastNoSeq  = 0b0010
	serialJSON     = 0b0001
	serialNone     = 0b0000
	compressionGzip = 0b0001
)

func buildFrame(msgType, flags, serialization byte, payload []byte) ([]byte, error) {
	var buf bytes.Buffer
	zw := gzip.NewWriter(&buf)
	if _, err := zw.Write(payload); err != nil {
		return nil, err
	}
	if err := zw.Close(); err != nil {
		return nil, err
	}
	head := []byte{
		0x11, // version 1 + header size 1
		msgType<<4 | flags,
		serialization<<4 | compressionGzip,
		0x00,
	}
	out := append(head, 0, 0, 0, 0)
	binary.BigEndian.PutUint32(out[4:8], uint32(buf.Len()))
	out = append(out, buf.Bytes()...)
	return out, nil
}

func BuildFullRequest(payloadJSON []byte) ([]byte, error) {
	return buildFrame(msgTypeFullRequest, flagNoSeq, serialJSON, payloadJSON)
}

func BuildAudioChunk(audio []byte, last bool) ([]byte, error) {
	flags := byte(flagNoSeq)
	if last {
		flags = flagLastNoSeq
	}
	return buildFrame(msgTypeAudioOnly, flags, serialNone, audio)
}

type ServerFrame struct {
	Type    int
	Last    bool
	Code    int64
	Payload []byte
}

// ParseServerFrame 解析单个完整帧（调用方按 header+size 自行切帧后传入）。
func ParseServerFrame(data []byte) (ServerFrame, error) {
	if len(data) < 8 {
		return ServerFrame{}, fmt.Errorf("帧过短: %d bytes", len(data))
	}
	if data[0]>>4 != 0b0001 {
		return ServerFrame{}, fmt.Errorf("不支持的协议版本: %x", data[0]>>4)
	}
	msgType := data[1] >> 4
	flags := data[1] & 0x0f
	compression := data[2] & 0x0f
	sf := ServerFrame{Type: int(msgType), Last: flags == flagLastNoSeq || flags == 0b0011}

	body := data[8:]
	switch msgType {
	case msgTypeFullResponse:
		if flags == 0b0001 || flags == 0b0011 {
			body = body[4:] // sequence
		}
		if len(body) < 4 {
			return ServerFrame{}, fmt.Errorf("响应帧过短")
		}
	case msgTypeError:
		if len(body) < 8 {
			return ServerFrame{}, fmt.Errorf("错误帧过短")
		}
		sf.Code = int64(binary.BigEndian.Uint32(body[:4]))
		body = body[8:]
	default:
		return ServerFrame{}, fmt.Errorf("未知消息类型: %x", msgType)
	}

	size := binary.BigEndian.Uint32(body[:4])
	raw := body[4:]
	if int(size) > len(raw) {
		return ServerFrame{}, fmt.Errorf("payload 长度不符: %d > %d", size, len(raw))
	}
	raw = raw[:size]
	if compression == compressionGzip {
		zr, err := gzip.NewReader(bytes.NewReader(raw))
		if err != nil {
			return ServerFrame{}, fmt.Errorf("gzip 解压失败: %w", err)
		}
		out, err := io.ReadAll(zr)
		if err != nil {
			return ServerFrame{}, fmt.Errorf("gzip 读取失败: %w", err)
		}
		raw = out
	}
	sf.Payload = raw
	return sf, nil
}
```
